4. Script to check file type

```
#!/bin/bash

read -p "Enter file or directory name: " file

if [ -f "$file" ]; then
    echo "$file is a regular file."
elif [ -d "$file" ]; then
    echo "$file is a directory."
else
    echo "$file is another type of file."
fi
```

6. Script Compare numbers

```
#!/bin/bash

num1=$1
num2=$2

if [ "$num1" -gt "$num2" ]; then
    echo "$num1 is greater than $num2"
elif [ "$num1" -lt "$num2" ]; then
    echo "$num1 is smaller than $num2"
else
    echo "$num1 is equal to $num2"
fi
```

7. Permission check

```
#!/bin/bash

file_path="$1"

if [ -e "$file_path" ]; then
    echo "$file_path passwords are enabled."
    if [ -w "$file_path" ]; then
        echo "You have permissions to edit $file_path."
    else
        echo "You do NOT have permissions to edit $file_path."
    fi
else
    echo "$file_path does not exist."
fi
``` 
